		tag = NOV
		NOT = { is_year = 1500 }

				set_province_flag = lit_peasant_faction_flag

			limit = {
				has_province_flag = lit_noble_faction_flag
			}
			clr_province_flag = lit_noble_faction_flag
			spawn_rebels = {
				type = noble_rebels
				size = 1
			}
		}
		random_owned_province = {
			limit = {
				has_province_flag = lit_peasant_faction_flag
			}
			clr_province_flag = lit_peasant_faction_flag
			spawn_rebels = {
				type = anti_tax_rebels
				size = 1
			}
		}
		hidden_effect = {
			random_owned_province = {
				limit = {
					has_province_flag = lit_merchant_faction_flag
				}
				clr_province_flag = lit_merchant_faction_flag
			}
		}
	}
	option = {
		name = "flavor_nov.EVTOPTB1"
		add_prestige = 5
		add_legitimacy = 5
		random_owned_province = {

				has_province_flag = lit_peasant_faction_flag

				limit = {
					has_province_flag = lit_merchant_faction_flag

					has_province_flag = lit_noble_faction_flag
				}

country_event = {
	id = flavor_nov.2
	title = flavor_nov.2.t
	desc = flavor_nov.2.d
	picture = DIPLOMACY_eventPicture
	is_triggered_only = yes

	option = {
		name = flavor_nov.2.a
		trigger = { tag = NOV }

		change_tag = RUS
		swap_non_generic_missions = yes
		on_change_tag_effect = yes

		if = {
			limit = {
				has_custom_ideas = no
				NOT = { has_idea_group = RUS_ideas }
			}
			country_event = { id = ideagroups.1 }
		}
		if = {
			if = {
				limit = { government = republic }
				add_government_reform = great_veche_republic
			}
			else = {
				if = {
					limit = { NOT = { government = monarchy } }
					change_government = monarchy
				}
				add_government_reform = tsardom
			}
		}
		RUS = { set_government_rank = 3 }
		russia_region = {
			limit = {
				NOT = { owned_by = ROOT }
			}
			add_permanent_claim = RUS
		}
		crimea_region = {
			limit = {
				NOT = { owned_by = ROOT }
			}
			add_permanent_claim = RUS
		}
		ural_region = {
			limit = {
				NOT = {
					owned_by = ROOT
				}
				is_empty = no
			}
			add_permanent_claim = RUS
		}
		add_country_modifier = {
			name = "centralization_modifier"
			duration = 7300
		}

		set_country_flag = formed_russia_flag
		add_prestige = 25
	}

	option = {
		name = flavor_nov.2.b
		add_prestige = 10
	}
}
