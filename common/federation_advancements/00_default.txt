federal_constitution = {
	gfx = GFX_federal_constitution
	can_pick = {
		# Need all other advancements
		num_federation_advancements = 7
		is_migratory_tribe = no
		is_year = 1575
	}

	effect = {
		# Save the membership before changing tag. Some tag-change effects (most
		# data before every_federation_member is evaluated later in this effect.
		hidden_effect = {
			every_federation_member = {
				limit = { ai = yes }
				set_country_flag = efv_federation_member_to_integrate
			}
		}

		# Switch Tag
		if = {
			limit = {
				has_government_attribute = efv_italian_federation_government
				NOT = { exists = ITA }
			}
			change_tag = ITA
			swap_non_generic_missions = yes
			add_country_modifier = { name = italian_federation duration = -1 }
			on_change_tag_effect = yes
			if = {
				limit = {
					has_custom_ideas = no
					NOT = { has_idea_group = ITA_ideas }
				}
				country_event = { id = ideagroups.1 }
			}
			set_country_flag = formed_italy_flag
		}
		else_if = { 
			limit = { primary_culture = creek NOT = { exists = CRE } }
			change_tag = CRE
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = CRE }
			if = {
				limit = {
					NOT = { has_idea_group = CRE_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else_if = { 
			limit = { primary_culture = dakota NOT = { exists = SIO } }
			change_tag = SIO
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = SIO }
			if = {
				limit = {
					NOT = { has_idea_group = siouan_federation_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else_if = { 
			limit = { primary_culture = huron NOT = { exists = HUR } }
			change_tag = HUR
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = HUR }
			if = {
				limit = {
					NOT = { has_idea_group = HUR_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else_if = { 
			limit = { primary_culture = illini NOT = { exists = ILL } }
			change_tag = ILL
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = ILL }
			if = {
				limit = {
					NOT = { has_idea_group = ne_native_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else_if = { 
			limit = { primary_culture = iroquois NOT = { exists = IRO } }
			change_tag = IRO
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = IRO }
			if = {
				limit = {
					NOT = { has_idea_group = IRO_ideas }
				}
				swap_federation_idea_group = yes
			}
		} 
		else_if = { 
			limit = { primary_culture = anishinabe NOT = { exists = POT } }
			change_tag = POT
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = POT }
			if = {
				limit = {
					NOT = { has_idea_group = ne_native_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else_if = { 
			limit = { primary_culture = shawnee NOT = { exists = SHA } }
			change_tag = SHA
			on_change_tag_effect = yes
			add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = SHA }
			if = {
				limit = {
					NOT = { has_idea_group = SHA_ideas }
				}
				swap_federation_idea_group = yes
			}
		}
		else = {
			form_federation_nation = {
				post_effect = {
					add_permanent_claims_for_federation_nation = { ORIGINAL = ROOT TAG = THIS }
					swap_federation_idea_group = yes
				}
			}
		}
		# Unite federation
		custom_tooltip = INTEGRATE_FEDERATION_MEMBERS
		hidden_effect = {
			every_country = {
				limit = {
					ai = yes
					has_country_flag = efv_federation_member_to_integrate
				}
				every_tribal_land_province = {
					limit = {
						OR = {
							owned_by = PREV
							is_empty = yes
						}
					}
					change_tribal_land = ROOT
				}
			}
			every_country = {
				limit = {
					ai = yes
					has_country_flag = efv_federation_member_to_integrate
				}
				every_tribal_land_province = {
					limit = {
						NOT = { owned_by = PREV }
					}
					change_tribal_land = ---
				}
				clr_country_flag = efv_federation_member_to_integrate
				ROOT = {
					inherit = PREV
				}
			}
			every_owned_province = {
				limit = {	#Kinda unnecessary, but better safe than sorry unless we want colonies to vanish once again
					OR = {
						owned_by = PREV
						is_empty = yes
					}
				}
				change_tribal_land = ROOT
			}

			# Record the Native buildings before removing native_mechanic.
			every_owned_province = {
				if = {
					limit = { has_building = native_earthwork }
					set_province_flag = efv_convert_native_earthwork
				}
				if = {
					limit = { has_building = native_palisade }
					set_province_flag = efv_convert_native_palisade
				}
				if = {
					limit = { has_building = native_fortified_house }
					set_province_flag = efv_convert_native_fortified_house
				}
				if = {
					limit = { has_building = native_ceremonial_fire_pit }
					set_province_flag = efv_convert_native_ceremonial_fire_pit
				}
				if = {
					limit = { has_building = native_irrigation }
					set_province_flag = efv_convert_native_irrigation
				}
				if = {
					limit = { has_building = native_storehouse }
					set_province_flag = efv_convert_native_storehouse
				}
				if = {
					limit = { has_building = native_longhouse }
					set_province_flag = efv_convert_native_longhouse
				}
				if = {
					limit = { has_building = native_sweat_lodge }
					set_province_flag = efv_convert_native_sweat_lodge
				}
				if = {
					limit = { has_building = native_great_trail }
					set_province_flag = efv_convert_native_great_trail
				}
				if = {
					limit = { has_building = native_three_sisters_field }
					set_province_flag = efv_convert_native_three_sisters_field
				}
			}
		}

		# Explicitly remove every reform that grants the Italian federation's
		# native_mechanic. Locked first-tier reforms are not reliably replaced
		# by change_government_via_reform on their own.
		if = {
			limit = { tag = ITA }
			if = {
				limit = { has_reform = signoria_reform }
				remove_government_reform = signoria_reform
			}
			if = {
				limit = { has_reform = venice_merchants_reform }
				remove_government_reform = venice_merchants_reform
			}
			if = {
				limit = { has_reform = ambrosian_republic }
				remove_government_reform = ambrosian_republic
			}
			if = {
				limit = { has_reform = papacy_reform }
				remove_government_reform = papacy_reform
			}
			if = {
				limit = { has_reform = efv_italian_federation_monarchy_reform }
				remove_government_reform = efv_italian_federation_monarchy_reform
			}

			if = {
				limit = { government = monarchy }
				change_government = monarchy
				add_government_reform = feudalism_reform
			}
			else_if = {
				limit = { government = republic }
				change_government = republic
				add_government_reform = merchants_reform
			}
			else_if = {
				limit = { government = theocracy }
				change_government = theocracy
					add_government_reform = leading_clergy_reform
				}

			# Native buildings cannot be replaced with ordinary buildings until
			# native_mechanic has been removed from the country.
			every_owned_province = {
				if = {
					limit = { has_province_flag = efv_convert_native_earthwork }
					remove_building = native_earthwork
					if = { limit = { NOT = { has_building = ramparts } } add_building = ramparts }
					clr_province_flag = efv_convert_native_earthwork
				}
				if = {
					limit = { has_province_flag = efv_convert_native_palisade }
					remove_building = native_palisade
					if = { limit = { NOT = { has_building = fort_15th } } add_building = fort_15th }
					clr_province_flag = efv_convert_native_palisade
				}
				if = {
					limit = { has_province_flag = efv_convert_native_fortified_house }
					remove_building = native_fortified_house
					if = { limit = { NOT = { has_building = barracks } } add_building = barracks }
					clr_province_flag = efv_convert_native_fortified_house
				}
				if = {
					limit = { has_province_flag = efv_convert_native_ceremonial_fire_pit }
					remove_building = native_ceremonial_fire_pit
					if = { limit = { NOT = { has_building = courthouse } } add_building = courthouse }
					clr_province_flag = efv_convert_native_ceremonial_fire_pit
				}
				if = {
					limit = { has_province_flag = efv_convert_native_irrigation }
					remove_building = native_irrigation
					if = { limit = { NOT = { has_building = workshop } } add_building = workshop }
					clr_province_flag = efv_convert_native_irrigation
				}
				if = {
					limit = { has_province_flag = efv_convert_native_storehouse }
					remove_building = native_storehouse
					if = { limit = { NOT = { has_building = marketplace } } add_building = marketplace }
					clr_province_flag = efv_convert_native_storehouse
				}
				if = {
					limit = { has_province_flag = efv_convert_native_longhouse }
					remove_building = native_longhouse
					if = { limit = { NOT = { has_building = town_hall } } add_building = town_hall }
					clr_province_flag = efv_convert_native_longhouse
				}
				if = {
					limit = { has_province_flag = efv_convert_native_sweat_lodge }
					remove_building = native_sweat_lodge
					if = { limit = { NOT = { has_building = university } } add_building = university }
					clr_province_flag = efv_convert_native_sweat_lodge
				}
				if = {
					limit = { has_province_flag = efv_convert_native_great_trail }
					remove_building = native_great_trail
					if = { limit = { NOT = { has_building = regimental_camp } } add_building = regimental_camp }
					clr_province_flag = efv_convert_native_great_trail
				}
				if = {
					limit = { has_province_flag = efv_convert_native_three_sisters_field }
					remove_building = native_three_sisters_field
					if = { limit = { NOT = { has_building = manufactory } } add_building = manufactory }
					clr_province_flag = efv_convert_native_three_sisters_field
				}
			}
		}
		
		if = {
			limit = {
				NOT = { government_rank = 2 }
			}
			set_government_rank_force = 2
		}
		add_country_modifier = {
			name = "centralization_modifier"
			duration = 7300
		}
		add_prestige = 25

		hidden_effect = {
			disband_federation = yes
		}
	}

	ai_chance = {
		factor = 1
		modifier = {
			factor = 0
			NOT = {
				has_government_attribute = efv_italian_federation_government
			}
		}
	}
}
